# ==========================================

def speak(text):

    print("Assistant:", text)

    try:

        with speech_lock:

            tts_engine.say(text)
            tts_engine.runAndWait()

    except Exception as e:

        print("TTS Error:", e)

def welcome_person():

# ==========================================
# ASK OLLAMA
# ==========================================

def ask_ollama_text(user_text):

    try:

        prompt = f"""
You are the voice assistant for CTC Innovation Lab.

The user said:
"{user_text}"

Give a short, friendly and useful answer.

Keep the answer under 40 words.

Do not make assumptions about the user.

Return only the answer that should be spoken.
"""

        data = {
            "model": LLM_MODEL,
            "prompt": prompt,
            "stream": False
        }

        response = requests.post(
            OLLAMA_URL,
            json=data,
            timeout=60
        )

        if response.status_code == 200:

            result = response.json()

            answer = result.get(
                "response",
                ""
            ).strip()

            if answer:

                speak(answer)

            else:

                speak(
                    "Sorry, I could not generate a response."
                )

        else:

            print(
                "Ollama error:",
                response.status_code
            )

            speak(
                "Sorry, I am having trouble connecting to my AI model."
            )

    except Exception as e:

        print(
            "Ollama connection error:",
            e
        )

        speak(
            "Sorry, I cannot connect to the AI server right now."
        )

# LISTEN TO MICROPHONE
# ==========================================

def listen_to_user():

    global voice_busy

    if microphone is None or vosk_model is None:

        print("No microphone or Vosk model available.")

        return

    voice_busy = True

    stream = None

    try:

        print("\nListening...")

        print("Speak now...")

        recognizer = KaldiRecognizer(
            vosk_model,
            VOSK_SAMPLE_RATE
        )

        recognizer.SetWords(False)

        stream = microphone
        stream.start_stream()

        start_time = time.time()
        speech_started = False
        text = ""

        while True:

            data = stream.read(
                VOSK_CHUNK_SIZE,
                exception_on_overflow=False
            )

            current_time = time.time()
            elapsed_time = current_time - start_time

            if recognizer.AcceptWaveform(data):

                result = json.loads(
                    recognizer.Result()
                )

                text = result.get(
                    "text",
                    ""
                ).strip()

                if text:

                    speech_started = True
                    break

            else:

                partial_result = json.loads(
                    recognizer.PartialResult()
                )

                partial_text = partial_result.get(
                    "partial",
                    ""
                ).strip()

                if partial_text:

                    speech_started = True

            # Wait up to 5 seconds for the user to start speaking.
            if not speech_started and elapsed_time >= 5:

                print("No speech detected.")

                return

            # Limit the total listening time to 8 seconds.
            if elapsed_time >= 8:

                break

        # ==================================
        # SPEECH TO TEXT
        # ==================================

        print(
            "Converting speech to text..."
        )

        if not text:

            final_result = json.loads(
                recognizer.FinalResult()
            )

            text = final_result.get(
                "text",
                ""
            ).strip()

        if text:

            print(
                "You said:",
                text
            )

        else:

            print(
                "Could not understand speech."
            )

            speak(
                "Sorry, I didn't understand that."
            )

            return

        # ==================================
        # SEND TO OLLAMA
        # ==================================

        ask_ollama_text(text)

    except Exception as e:

        print(
            "Microphone error:",
            e
        )

    finally:

        if stream is not None:

            try:

                stream.stop_stream()

            except Exception:

                pass

        voice_busy = False



# ==========================================

def start_voice_assistant():

    global voice_busy

    if voice_busy:

        return

    voice_thread = threading.Thread(
        target=listen_to_user,
        daemon=True
    )

    voice_thread.start()

# CAMERA DETECTION LOOP
# ==========================================

def camera_detection_loop(
    screen_width,
    screen_height
):

    global person_detected
    global previous_person_detected
    global is_running
    global current_surface
    global cached_boxes
    global last_person_left_time

    cap = cv2.VideoCapture(
        CAMERA_INDEX
    )

    if not cap.isOpened():

        print(
            f"Error: Could not open camera "
            f"at index {CAMERA_INDEX}"
        )

        return

    # ======================================
    # CAMERA SETTINGS
    # ======================================

    cap.set(
        cv2.CAP_PROP_FRAME_WIDTH,
        CAM_WIDTH
    )

    cap.set(
        cv2.CAP_PROP_FRAME_HEIGHT,
        CAM_HEIGHT
    )

    cap.set(
        cv2.CAP_PROP_FPS,
        30
    )

    print(
        "Camera Thread Started..."
    )

    frame_count = 0

    # ======================================
    # CAMERA LOOP
    # ======================================

    while is_running:

        ret, frame = cap.read()

        if not ret:

            continue

        frame_count += 1

        h, w, _ = frame.shape

        center_min_x = int(
            w * CENTER_MIN_RATIO
        )

        center_max_x = int(
            w * CENTER_MAX_RATIO
        )

        # ==================================
        # RUN YOLO
        # ==================================

        if frame_count % YOLO_SKIP_FRAMES == 0:

            cached_boxes.clear()

            detected_in_center = False

            results = yolo_model(
                frame,
                verbose=False,
                conf=CONFIDENCE_THRESHOLD,
                classes=[0],
                imgsz=320
            )

            # ==================================
            # PROCESS DETECTIONS
            # ==================================

            for result in results:

                for box in result.boxes:

                    x1, y1, x2, y2 = map(
                        int,
                        box.xyxy[0].tolist()
                    )

                    person_center_x = (
                        x1 + x2
                    ) / 2.0

                    is_in_center = (
                        center_min_x
                        <= person_center_x
                        <= center_max_x
                    )

                    if is_in_center:

                        detected_in_center = True

                    cached_boxes.append(
                        (
                            x1,
                            y1,
                            x2,
                            y2,
                            is_in_center
                        )
                    )

            # ==================================
            # UPDATE PERSON STATUS
            # ==================================

            person_detected = (
                detected_in_center
            )

            # ==================================
            # PERSON ENTERED CENTER
            # ==================================

            if (
                person_detected
                and not previous_person_detected
            ):

                current_time = time.time()

                time_since_last_person = (
                    current_time
                    - last_person_left_time
                )

                # ==================================
                # CHECK 4 SECOND DELAY
                # ==================================

                if (
                    time_since_last_person
                    >= WELCOME_DELAY
                ):

                    print(
                        "Person entered center zone!"
                    )

                    print(
                        "Saying welcome..."
                    )

                    # ------------------------------
                    # WELCOME MESSAGE
                    # ------------------------------

                    threading.Thread(
                        target=welcome_person,
                        daemon=True
                    ).start()

                    # ------------------------------
                    # START MICROPHONE
                    # ------------------------------

                    if MIC_ENABLED:

                        def delayed_listen():

                            # Wait for welcome
                            # message to finish

                            time.sleep(3)

                            start_voice_assistant()

                        threading.Thread(
                            target=delayed_listen,
                            daemon=True
                        ).start()

                else:

                    remaining = (
                        WELCOME_DELAY
                        - time_since_last_person
                    )

                    print(
                        f"Waiting {remaining:.1f} "
                        f"seconds before welcoming."
                    )

            # ==================================
            # PERSON LEFT CENTER
            # ==================================

            if (
                not person_detected
                and previous_person_detected
            ):

                print(
                    "Person left center zone."
                )

                # Record exact time person left

                last_person_left_time = (
                    time.time()
                )

                print(
                    "4 second welcome cooldown started."
                )

            # ==================================
            # SAVE CURRENT STATE
            # ==================================

            previous_person_detected = (
                person_detected
            )

        # ==================================
        # DRAW BOUNDING BOXES
        # ==================================

        for (
            x1,
            y1,
            x2,
            y2,
            is_in_center
        ) in cached_boxes:

            if is_in_center:

                color = (
                    0,
                    255,
                    0
                )

            else:

                color = (
                    0,
                    165,
                    255
                )

            cv2.rectangle(
                frame,
                (x1, y1),
                (x2, y2),
                color,
                2
            )

            cv2.putText(
                frame,
                "Person",
                (x1, y1 - 10),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.5,
                color,
                2
            )

        # ==================================
        # CENTER ZONE LINES
        # ==================================

        cv2.line(
            frame,
            (center_min_x, 0),
            (center_min_x, h),
            (255, 255, 255),
            1
        )

        cv2.line(
            frame,
            (center_max_x, 0),
            (center_max_x, h),
            (255, 255, 255),
            1
        )

        # ==================================
        # PYGAME FRAME
        # ==================================

        resized_frame = cv2.resize(
            frame,
            (
                screen_width,
                screen_height
            ),
            interpolation=cv2.INTER_NEAREST
        )

        rgb_frame = cv2.cvtColor(
            resized_frame,
            cv2.COLOR_BGR2RGB
        )

        current_surface = pygame.image.frombuffer(
            rgb_frame.tobytes(),
            (
                screen_width,
                screen_height
            ),
            "RGB"
        )

    # ======================================
    # CAMERA SHUTDOWN
    # ======================================

    cap.release()

    print(
        "Camera thread shut down cleanly."
    )

def draw_wrapped_text(surface, text, x, y, max_width, font, color, line_spacing=4):
    words = text.split(" ")
    lines = []
    current_line = []

    for word in words:
        test_line = " ".join(current_line + [word])
        if font.size(test_line)[0] <= max_width:
            current_line.append(word)
        else:
            if current_line:
                lines.append(" ".join(current_line))
            current_line = [word]

    if current_line:
        lines.append(" ".join(current_line))

    curr_y = y
    for line in lines:
        line_surf = font.render(line, True, color)
        surface.blit(line_surf, (x, curr_y))
        curr_y += line_surf.get_height() + line_spacing

    return curr_y
