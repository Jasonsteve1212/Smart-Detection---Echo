import sys
import time
import threading
import cv2
import pygame
import requests
import pyttsx3
import speech_recognition as sr
from ultralytics import YOLO

# ==========================================
# 1. IMMEDIATE PYGAME INITIALIZATION
# ==========================================
# Initialize Pygame first so the window opens without waiting for AI/Audio
pygame.init()
info = pygame.display.Info()
WIDTH, HEIGHT = info.current_w, info.current_h

screen = pygame.display.set_mode((WIDTH, HEIGHT), pygame.FULLSCREEN | pygame.DOUBLEBUF)
pygame.display.set_caption("CTC Innovation Lab - AI Smart Mirror")
clock = pygame.time.Clock()

# Fonts & Colors
WHITE  = (255, 255, 255)
CYAN   = (0, 225, 255)
GREEN  = (0, 255, 128)
GRAY   = (180, 180, 180)
YELLOW = (255, 215, 0)
PURPLE = (180, 130, 255)

font_clock  = pygame.font.SysFont("Arial", 54, bold=True)
font_temp   = pygame.font.SysFont("Arial", 72, bold=True)
font_medium = pygame.font.SysFont("Arial", 28, bold=True)
font_body   = pygame.font.SysFont("Arial", 22)
font_small  = pygame.font.SysFont("Arial", 18)

# ==========================================
# CONFIGURATION
# ==========================================
CAMERA_INDEX = 0
CONFIDENCE_THRESHOLD = 0.5
CAM_WIDTH, CAM_HEIGHT = 640, 480
YOLO_SKIP_FRAMES = 3

OLLAMA_URL = "http://localhost:11434/api/generate"
LLM_MODEL = "gemma3:4b"

WELCOME_MESSAGE = "Welcome to CTC Innovation Lab!"
WELCOME_DELAY = 4
MIC_ENABLED = True

CITY = "Coimbatore"
CONSTANT_TEMP = "28°C"
CONSTANT_DESC = "Partly Cloudy"
CONSTANT_HUMIDITY = "Humidity: 65%"

CENTER_MIN_RATIO = 0.3
CENTER_MAX_RATIO = 0.7

# ==========================================
# GLOBAL STATE & LOCKS
# ==========================================
is_running = True
person_detected = False
previous_person_detected = False
last_person_left_time = -WELCOME_DELAY

latest_raw_frame = None
frame_lock = threading.Lock()
cached_boxes = []

voice_busy = False
speech_lock = threading.Lock()
voice_state = "idle"  # "idle", "listening", "processing", "speaking"
user_spoken_text = ""
assistant_reply_text = ""
conversation_history = []
MAX_HISTORY = 6

# ==========================================
# AI & HARDWARE INITIALIZATION
# ==========================================
print("Loading YOLO model...")
yolo_model = YOLO("yolov8n.pt")

print("Initializing TTS...")
tts_engine = pyttsx3.init()
tts_engine.setProperty("rate", 170)
tts_engine.setProperty("volume", 1.0)

recognizer = sr.Recognizer()
recognizer.energy_threshold = 300
recognizer.dynamic_energy_threshold = True
recognizer.pause_threshold = 0.8

microphone = None
try:
    microphone = sr.Microphone()
    print("Microphone initialized.")
except Exception as e:
    print(f"Microphone initialization warning: {e}")

# ==========================================
# TEXT-TO-SPEECH
# ==========================================
def speak(text):
    global voice_state, assistant_reply_text
    print("Assistant:", text)
    assistant_reply_text = text
    voice_state = "speaking"

    try:
        with speech_lock:
            tts_engine.say(text)
            tts_engine.runAndWait()
    except Exception as e:
        print("TTS Error:", e)
    finally:
        voice_state = "idle"

def welcome_person():
    speak(WELCOME_MESSAGE)

# ==========================================
# OLLAMA LLM CONVERSATION
# ==========================================
def ask_ollama_text(user_text):
    global voice_state, conversation_history
    voice_state = "processing"

    conversation_history.append({"role": "user", "content": user_text})
    if len(conversation_history) > MAX_HISTORY:
        conversation_history = conversation_history[-MAX_HISTORY:]

    system_prompt = (
        "You are the voice assistant for CTC Innovation Lab.\n"
        "Give short, friendly, conversational answers under 35 words.\n"
        "Return only the direct response to be spoken."
    )

    formatted_dialogue = ""
    for msg in conversation_history:
        formatted_dialogue += f"{msg['role'].capitalize()}: {msg['content']}\n"

    full_prompt = f"{system_prompt}\n\n{formatted_dialogue}Assistant:"

    try:
        response = requests.post(
            OLLAMA_URL,
            json={"model": LLM_MODEL, "prompt": full_prompt, "stream": False},
            timeout=15
        )
        if response.status_code == 200:
            answer = response.json().get("response", "").strip()
            if not answer:
                answer = "I'm not sure how to answer that."
        else:
            answer = "I am having trouble connecting to my AI model."
    except Exception as e:
        print("Ollama Error:", e)
        answer = "I cannot connect to the AI server right now."

    conversation_history.append({"role": "assistant", "content": answer})
    speak(answer)

# ==========================================
# CONTINUOUS VOICE LISTENING LOOP
# ==========================================
def continuous_conversation_session():
    global voice_busy, voice_state, user_spoken_text, person_detected, conversation_history

    if microphone is None:
        return

    voice_busy = True
    consecutive_silence = 0
    MAX_SILENCE = 2

    print("\n[Voice Session Active]")

    while person_detected and is_running:
        voice_state = "listening"
        print("🎙️ Listening...")

        audio_data = None
        try:
            with microphone as source:
                audio_data = recognizer.listen(source, timeout=5, phrase_time_limit=8)
        except sr.WaitTimeoutError:
            consecutive_silence += 1
            print(f"Silence timeout ({consecutive_silence}/{MAX_SILENCE})")
            if consecutive_silence >= MAX_SILENCE:
                break
            continue
        except Exception as e:
            print(f"Mic capture error: {e}")
            break

        if audio_data is not None:
            voice_state = "processing"
            try:
                print("Processing speech...")
                user_text = recognizer.recognize_google(audio_data).strip()
                print(f"User: {user_text}")

                if not user_text:
                    continue

                user_spoken_text = user_text
                consecutive_silence = 0

                if any(k in user_text.lower() for k in ["bye", "goodbye", "exit", "quit", "stop"]):
                    speak("Goodbye! Have a great day.")
                    break

                ask_ollama_text(user_text)

            except sr.UnknownValueError:
                consecutive_silence += 1
                speak("Sorry, I didn't catch that.")
                if consecutive_silence >= MAX_SILENCE:
                    break
            except sr.RequestError as e:
                print(f"STT Service error: {e}")
                speak("Speech recognition service unavailable.")
                break

    print("[Voice Session Ended]")
    conversation_history.clear()
    voice_state = "idle"
    voice_busy = False

def start_voice_assistant():
    global voice_busy
    if voice_busy:
        return
    threading.Thread(target=continuous_conversation_session, daemon=True).start()

# ==========================================
# CAMERA DETECTION THREAD
# ==========================================
def camera_detection_loop(screen_width, screen_height):
    global person_detected, previous_person_detected, is_running
    global latest_raw_frame, cached_boxes, last_person_left_time

    cap = cv2.VideoCapture(CAMERA_INDEX)
    if not cap.isOpened():
        print(f"Error: Could not open camera at index {CAMERA_INDEX}")
        return

    cap.set(cv2.CAP_PROP_FRAME_WIDTH, CAM_WIDTH)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, CAM_HEIGHT)
    cap.set(cv2.CAP_PROP_FPS, 30)

    print("Camera Thread Started...")
    frame_count = 0

    while is_running:
        ret, frame = cap.read()
        if not ret:
            time.sleep(0.01)
            continue

        frame_count += 1
        h, w, _ = frame.shape
        center_min_x = int(w * CENTER_MIN_RATIO)
        center_max_x = int(w * CENTER_MAX_RATIO)

        if frame_count % YOLO_SKIP_FRAMES == 0:
            cached_boxes.clear()
            detected_in_center = False

            results = yolo_model(
                frame,
                verbose=False,
                conf=CONFIDENCE_THRESHOLD,
                classes=[0],
                imgsz=320
            )

            for result in results:
                for box in result.boxes:
                    x1, y1, x2, y2 = map(int, box.xyxy[0].tolist())
                    person_center_x = (x1 + x2) / 2.0
                    is_in_center = center_min_x <= person_center_x <= center_max_x

                    if is_in_center:
                        detected_in_center = True

                    cached_boxes.append((x1, y1, x2, y2, is_in_center))

            person_detected = detected_in_center

            # Welcome trigger & auto start follow-up conversation
            if person_detected and not previous_person_detected:
                current_time = time.time()
                time_since_last_person = current_time - last_person_left_time

                if time_since_last_person >= WELCOME_DELAY:
                    print("Person entered center zone! Greeting...")
                    
                    def welcome_and_start_chat():
                        welcome_person()
                        if MIC_ENABLED:
                            start_voice_assistant()

                    threading.Thread(target=welcome_and_start_chat, daemon=True).start()

            if not person_detected and previous_person_detected:
                print("Person left center zone.")
                last_person_left_time = time.time()

            previous_person_detected = person_detected

        # Annotate bounding boxes
        for (x1, y1, x2, y2, is_in_center) in cached_boxes:
            color = (0, 255, 0) if is_in_center else (0, 165, 255)
            cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
            cv2.putText(frame, "Person", (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

        # Center Zone boundary lines
        cv2.line(frame, (center_min_x, 0), (center_min_x, h), (255, 255, 255), 1)
        cv2.line(frame, (center_max_x, 0), (center_max_x, h), (255, 255, 255), 1)

        # Prepare frame for Pygame safely
        resized = cv2.resize(frame, (screen_width, screen_height), interpolation=cv2.INTER_NEAREST)
        rgb_frame = cv2.cvtColor(resized, cv2.COLOR_BGR2RGB)

        with frame_lock:
            latest_raw_frame = rgb_frame.tobytes()

    cap.release()
    print("Camera thread shut down cleanly.")

# ==========================================
# UI HELPERS & OVERLAY
# ==========================================
city_surf = font_medium.render(f"📍 {CITY}", True, CYAN)
temp_surf = font_temp.render(CONSTANT_TEMP, True, WHITE)
desc_surf = font_medium.render(CONSTANT_DESC, True, GREEN)
humid_surf = font_body.render(CONSTANT_HUMIDITY, True, GRAY)

status_active = font_body.render("🟢 User In Center Zone", True, GREEN)
status_idle   = font_body.render("⚪ Standing By...", True, GRAY)

sidebar_width = 380
sidebar_surface = pygame.Surface((sidebar_width, HEIGHT), pygame.SRCALPHA)
sidebar_surface.fill((0, 0, 0, 160))

def draw_wrapped_text(surface, text, x, y, max_width, font, color, line_spacing=4):
    words = text.split(" ")
    lines = []
    current_line = []

    for word in words:
        test_line = " ".join(current_line + [word])
        if font.size(test_line)[0] <= max_width:
            current_line.append(word)
        else:
            if current_line:
                lines.append(" ".join(current_line))
            current_line = [word]

    if current_line:
        lines.append(" ".join(current_line))

    curr_y = y
    for line in lines:
        line_surf = font.render(line, True, color)
        surface.blit(line_surf, (x, curr_y))
        curr_y += line_surf.get_height() + line_spacing

    return curr_y

# ==========================================
# START BACKGROUND THREADS
# ==========================================
cam_thread = threading.Thread(
    target=camera_detection_loop,
    args=(WIDTH, HEIGHT),
    daemon=True
)
cam_thread.start()

# ==========================================
# MAIN DISPLAY LOOP
# ==========================================
running = True
last_time_str = ""
clock_surf = font_clock.render("00:00:00 AM", True, WHITE)
date_surf = font_body.render("Monday, Jan 01", True, GRAY)

while running:
    # 1. Event Handling
    for event in pygame.event.get():
        if event.type == pygame.QUIT or (event.type == pygame.KEYDOWN and event.key == pygame.K_x):
            running = False

    # 2. Camera Frame Blit (Thread-Safe on Main Thread)
    raw_bytes = None
    with frame_lock:
        if latest_raw_frame is not None:
            raw_bytes = latest_raw_frame

    if raw_bytes is not None:
        cam_surface = pygame.image.frombuffer(raw_bytes, (WIDTH, HEIGHT), "RGB")
        screen.blit(cam_surface, (0, 0))
    else:
        screen.fill((20, 20, 20))

    # 3. Sidebar UI
    screen.blit(sidebar_surface, (0, 0))
    padding_x = 30

    now_str = time.strftime("%I:%M:%S %p")
    if now_str != last_time_str:
        last_time_str = now_str
        clock_surf = font_clock.render(now_str, True, WHITE)
        date_surf = font_body.render(time.strftime("%A, %b %d"), True, GRAY)

    screen.blit(clock_surf, (padding_x, 40))
    screen.blit(date_surf, (padding_x, 105))

    pygame.draw.line(screen, (255, 255, 255, 100), (padding_x, 140), (sidebar_width - padding_x, 140), 1)

    screen.blit(city_surf, (padding_x, 160))
    screen.blit(temp_surf, (padding_x, 205))
    screen.blit(desc_surf, (padding_x, 285))
    screen.blit(humid_surf, (padding_x, 325))

    # 4. Conversation Cards & State
    card_y = 375
    card_width = sidebar_width - (padding_x * 2)

    if voice_state == "listening":
        state_header = font_body.render("🎙️ Listening...", True, YELLOW)
    elif voice_state == "processing":
        state_header = font_body.render("🧠 Thinking...", True, PURPLE)
    elif voice_state == "speaking":
        state_header = font_body.render("🔊 Speaking...", True, CYAN)
    else:
        state_header = None

    if state_header:
        screen.blit(state_header, (padding_x, card_y))
        card_y += 35

    if user_spoken_text:
        screen.blit(font_small.render("YOU SAID:", True, GRAY), (padding_x, card_y))
        card_y += 20
        card_y = draw_wrapped_text(screen, f'"{user_spoken_text}"', padding_x, card_y, card_width, font_body, WHITE)
        card_y += 12

    if assistant_reply_text:
        screen.blit(font_small.render("AI ASSISTANT:", True, GRAY), (padding_x, card_y))
        card_y += 20
        card_y = draw_wrapped_text(screen, assistant_reply_text, padding_x, card_y, card_width, font_body, CYAN)

    status_surf = status_active if person_detected else status_idle
    screen.blit(status_surf, (padding_x, HEIGHT - 50))

    pygame.display.flip()
    clock.tick(60)

# ==========================================
# CLEAN SHUTDOWN
# ==========================================
print("Shutting down Smart Mirror...")
is_running = False
cam_thread.join(timeout=2.0)
pygame.quit()
sys.exit(0)
